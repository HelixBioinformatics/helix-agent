"""Optional bounded FreeSASA observations. No burial cutoff or fold classifier."""
import hashlib
import importlib.metadata
import json
import math
from pathlib import Path
import statistics
import sys
from tempfile import TemporaryDirectory

VERSION = 'helix-surface-diagnostic/1.0.0'
SIDE = {'ALA': 'CB', 'ARG': 'CB CG CD NE CZ NH1 NH2', 'ASN': 'CB CG OD1 ND2',
        'ASP': 'CB CG OD1 OD2', 'CYS': 'CB SG', 'GLN': 'CB CG CD OE1 NE2',
        'GLU': 'CB CG CD OE1 OE2', 'GLY': '', 'HIS': 'CB CG ND1 CD2 CE1 NE2',
        'ILE': 'CB CG1 CG2 CD1', 'LEU': 'CB CG CD1 CD2', 'LYS': 'CB CG CD CE NZ',
        'MET': 'CB CG SD CE', 'PHE': 'CB CG CD1 CD2 CE1 CE2 CZ', 'PRO': 'CB CG CD',
        'SER': 'CB OG', 'THR': 'CB OG1 CG2', 'TRP': 'CB CG CD1 CD2 NE1 CE2 CE3 CZ2 CZ3 CH2',
        'TYR': 'CB CG CD1 CD2 CE1 CE2 CZ OH', 'VAL': 'CB CG1 CG2'}
CHARGED = {'ASP': 100.47, 'GLU': 132.18, 'LYS': 162.98, 'ARG': 196.17}
PARAMETERS = {'algorithm': 'LeeRichards', 'probe-radius': 1.4, 'n-slices': 20, 'n-points': 100, 'n-threads': 1}
OPTIONS = {'hetatm': False, 'hydrogen': False, 'join-models': False, 'skip-unknown': False, 'halt-at-unknown': True}


def digest(data):
    return hashlib.sha256(data).hexdigest()


def measure(request):
    import freesasa
    import gemmi
    versions = {name: importlib.metadata.version(name) for name in ('freesasa', 'gemmi')}
    if versions != {'freesasa': '2.2.1', 'gemmi': '0.7.5'}:
        raise ValueError('unsupported-engine-version')
    source, options = request['structure'], request['options']
    if len(source.encode('utf-8')) > 20_000_000:
        raise ValueError('structure-size')
    structure = (gemmi.make_structure_from_block(gemmi.cif.read_string(source).sole_block())
                 if options['format'] == 'mmcif' else gemmi.read_pdb_string(source))
    ordinal = options['model']
    if not isinstance(ordinal, int) or not 1 <= ordinal <= len(structure):
        raise ValueError('model-selection')
    chains = [c for c in structure[ordinal - 1] if c.name == options['chain']]
    if len(chains) != 1 or len(chains[0].name) > 1:
        raise ValueError('ambiguous-or-unsupported-author-chain')
    residues = [r for r in chains[0] if r.entity_type == gemmi.EntityType.Polymer or r.het_flag == 'A']
    if not residues or any(r.name not in SIDE for r in residues):
        raise ValueError('unsupported-polymer-residue')
    atoms = [a for r in residues for a in r]
    if len(atoms) > 30_000:
        raise ValueError('atom-limit')
    if any(a.altloc not in ('\x00', ' ') or abs(a.occ - 1) > 0.001 for a in atoms):
        raise ValueError('ambiguous-or-partially-occupied-atoms')
    if any(not all(math.isfinite(x) for x in (a.pos.x, a.pos.y, a.pos.z)) for a in atoms):
        raise ValueError('nonfinite-coordinates')
    atom_keys = [(str(r.seqid), a.name) for r in residues for a in r]
    if len(atom_keys) != len(set(atom_keys)):
        raise ValueError('duplicate-atom-identity')
    heavy = [a for a in atoms if not a.is_hydrogen()]
    if any(a.element.name not in ('C', 'N', 'O', 'S') for a in heavy):
        raise ValueError('unsupported-element')
    classifier = freesasa.Classifier.getStandardClassifier('protor')
    for r in residues:
        for a in r:
            if not a.is_hydrogen() and (classifier.radius(r.name, a.name) <= 0 or classifier.classify(r.name, a.name) not in ('Polar', 'Apolar')):
                raise ValueError('unknown-protor-atom')
    selected = gemmi.Structure()
    model, chain = gemmi.Model('1'), gemmi.Chain(options['chain'])
    for r in residues:
        chain.add_residue(r.clone())
    model.add_chain(chain)
    selected.add_model(model)
    coordinates = selected.make_pdb_string()
    with TemporaryDirectory(prefix='helix-surface-') as directory:
        file = Path(directory) / 'selected.pdb'
        file.write_text(coordinates)
        parsed = freesasa.Structure(str(file), classifier=classifier, options=OPTIONS)
        if parsed.nAtoms() != len(heavy):
            raise ValueError('parser-atom-inventory')
        expected = [(str(r.seqid), r.name, a.name, (a.pos.x, a.pos.y, a.pos.z)) for r in residues for a in r if not a.is_hydrogen()]
        for i, (resid, name, atom, xyz) in enumerate(expected):
            if (parsed.chainLabel(i), parsed.residueNumber(i).strip(), parsed.residueName(i).strip(), parsed.atomName(i).strip()) != (options['chain'], resid, name, atom):
                raise ValueError('parser-atom-inventory')
            # Gemmi's PDB serialization retains three coordinate decimals.
            if any(abs(x - y) > 0.00051 for x, y in zip(parsed.coord(i), xyz)):
                raise ValueError('parser-coordinate-precision')
        result = freesasa.calc(parsed, freesasa.Parameters(PARAMETERS))
        areas = result.residueAreas()
        if sum(len(v) for v in areas.values()) != len(residues):
            raise ValueError('parser-residue-inventory')
        totals = {name: sum(getattr(area, name) for values in areas.values() for area in values.values()) for name in ('total', 'mainChain', 'sideChain', 'polar', 'apolar')}
        if any(not math.isfinite(v) or v < 0 for v in totals.values()) or totals['total'] <= 0:
            raise ValueError('nonfinite-surface-area')
        if not math.isclose(totals['total'], result.totalArea(), rel_tol=1e-10, abs_tol=1e-7):
            raise ValueError('surface-area-identity')
        for parts in (('mainChain', 'sideChain'), ('polar', 'apolar')):
            if not math.isclose(sum(totals[p] for p in parts), totals['total'], rel_tol=1e-10, abs_tol=1e-7):
                raise ValueError('surface-area-identity')
        charged, missing_all = [], []
        for residue in residues:
            names = {a.name for a in residue if not a.is_hydrogen()}
            missing = sorted(set(('N CA C O ' + SIDE[residue.name]).split()) - names)
            if missing:
                missing_all.append({'residue': str(residue.seqid), 'residueName': residue.name, 'missingExpectedHeavyAtoms': missing})
            if residue.name not in CHARGED:
                continue
            area = areas[options['chain']][str(residue.seqid)]
            if not area.hasRelativeAreas or not math.isfinite(area.relativeSideChain) or area.relativeSideChain < 0:
                raise ValueError('relative-area-reference')
            if not math.isclose(area.relativeSideChain * CHARGED[residue.name], area.sideChain, rel_tol=1e-10, abs_tol=1e-7):
                raise ValueError('relative-area-reference')
            missing_side = sorted(set(SIDE[residue.name].split()) - names)
            charged.append({'residue': str(residue.seqid), 'residueName': residue.name,
                            'sidechainAreaAngstromSquared': area.sideChain, 'relativeSidechainArea': area.relativeSideChain,
                            'completeSidechain': not missing_side, 'missingSidechainAtoms': missing_side,
                            'interpretableRelativeSidechainArea': area.relativeSideChain if not missing_side else None})
    values = [r['interpretableRelativeSidechainArea'] for r in charged if r['completeSidechain']]
    gemmi_binary = list(Path(gemmi.__file__).parent.glob('gemmi_ext*.so'))
    if len(gemmi_binary) != 1:
        raise ValueError('engine-identity')
    modules = {'freesasa': digest(Path(freesasa.__file__).read_bytes()), 'gemmi': digest(Path(gemmi.__file__).read_bytes()),
               'gemmi_ext': digest(gemmi_binary[0].read_bytes())}
    return {'version': VERSION, 'status': 'measured', 'sourceSha256': digest(source.encode('utf-8')),
            'selectedCoordinatesSha256': digest(coordinates.encode('utf-8')),
            'scope': {'authorChain': options['chain'], 'modelOrdinal': ordinal, 'residues': len(residues),
                      'inputAtoms': len(atoms), 'heavyAtoms': len(heavy), 'excludedHydrogens': len(atoms) - len(heavy),
                      'selection': 'complete selected polymer chain; other chains, ligands and waters excluded',
                      'coordinatePreparation': 'Gemmi selected-chain PDB serialization, coordinates rounded to 0.001 angstrom'},
            'engine': {'name': 'FreeSASA Python binding', 'versions': versions, 'modulesSha256': modules,
                       'sourceConfigVersionText': '2.0.2', 'versionNote': 'Python distribution 2.2.1; bundled config.h text 2.0.2. Installed modules are identified by hashes.'},
            'parameters': PARAMETERS, 'classifier': 'protor',
            'reference': {'convention': 'extended Ala-X-Ala ProtOr sidechain area', 'sidechainAreasAngstromSquared': CHARGED, 'ratioCanExceedOne': True},
            'measurement': {'areaAngstromSquared': totals, 'apolarAreaFraction': totals['apolar'] / totals['total'],
                            'polarAreaFraction': totals['polar'] / totals['total'], 'chargedResidues': len(charged),
                            'completeChargedSidechains': len(values), 'incompleteChargedSidechains': len(charged) - len(values),
                            'chargedSidechainRelativeAreaMedian': statistics.median(values) if values else None,
                            'chargedSidechainRelativeAreaMin': min(values) if values else None,
                            'chargedSidechainRelativeAreaMax': max(values) if values else None},
            'chargedResidueObservations': charged, 'residuesWithMissingExpectedHeavyAtoms': missing_all,
            'foldVerdict': 'not_established', 'burialThreshold': None, 'confidenceScoresUsed': False,
            'limitations': ['Surface exposure alone does not establish fold validity, energetic stability, charge compensation or protonation.',
                            'Partial sidechains remain explicit and are excluded from interpretable relative-area summaries.',
                            'Missing neighboring atoms and omitted assembly partners can change the areas.',
                            'No calibrated burial cutoff or independent physical labels are supplied.']}


def main():
    raw = sys.stdin.buffer.read(40_000_001)
    if len(raw) > 40_000_000:
        raise ValueError('request-size')
    request = json.loads(raw)
    try:
        result = measure(request)
    except (ImportError, FileNotFoundError):
        result = {'version': VERSION, 'status': 'not_evaluable', 'failureCode': 'missing-validator-dependencies'}
    except ValueError as error:
        known = {'unsupported-engine-version', 'structure-size', 'model-selection', 'ambiguous-or-unsupported-author-chain',
                 'unsupported-polymer-residue', 'atom-limit', 'ambiguous-or-partially-occupied-atoms', 'nonfinite-coordinates',
                 'duplicate-atom-identity', 'unsupported-element', 'unknown-protor-atom', 'parser-atom-inventory',
                 'parser-coordinate-precision', 'parser-residue-inventory', 'nonfinite-surface-area', 'surface-area-identity',
                 'relative-area-reference', 'engine-identity'}
        result = {'version': VERSION, 'status': 'not_evaluable', 'failureCode': str(error) if str(error) in known else 'validator-failed'}
    except Exception:
        result = {'version': VERSION, 'status': 'not_evaluable', 'failureCode': 'validator-failed'}
    print(json.dumps(result, sort_keys=True, allow_nan=False))


if __name__ == '__main__':
    main()
